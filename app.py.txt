# 1. OPEN BEAUTY FACTS API INTEGRATION (Free, Open-Source Database)
# -----------------------------------------------------------------------------
def fetch_product_ingredients(product_name):
    """Searches Open Beauty Facts for a product and returns its ingredient list."""
    url = f"https://world.openbeautyfacts.org/cgi/search.pl?search_terms={product_name}&search_simple=1&action=process&json=1"
    headers = {"User-Agent": "MonadWatchdogApp - Educational/Research - v1.0"}
    
    try:
        response = requests.get(url, headers=headers, timeout=5)
        if response.status_status == 200:
            data = response.json()
            products = data.get("products", [])
            if products:
                first_match = products[0]
                name = first_match.get("product_name", product_name)
                ingredients_text = first_match.get("ingredients_text", None)
                return name, ingredients_text
    except Exception:
        pass
    return None, None
